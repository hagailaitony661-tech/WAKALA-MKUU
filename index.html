function processAutoSMS(){
  const urlParams = new URLSearchParams(window.location.search);
  const from = urlParams.get('from');
  const sms = urlParams.get('sms');

  if(sms){
    let account = 'HaloPesa';
    if(from && from.toUpperCase().includes('HALOTEL')) account = 'HaloPesa';
   
    let type = 'Deposit';
    let smsUpper = sms.toUpperCase();
    if(smsUpper.includes('UMETUMA') || smsUpper.includes('UMEHAMISHA') || smsUpper.includes('PAID') || smsUpper.includes('WITHDRAW')){
      type = 'Withdrawal';
    }

    let amountMatch = sms.match(/(?:TZS|TSh|Kiasi|Sh)\s*([\d,]+(?:\.\d+)?)/i) || sms.match(/([\d,]+(?:\.\d+)?)\s*(?:TZS|TSh)/i);
    let amount = amountMatch ? parseFloat(amountMatch[1].replace(/,/g, '')) : 0;

    let refMatch = sms.match(/(?:Txn ID|Kumbukumbu|Ref|ID|Transaction ID)[:\s]*([A-Z0-9]+)/i);
    let ref = refMatch ? refMatch[1] : 'AUTO-' + Date.now().toString().slice(-6);

    if(amount > 0){
      let isDuplicate = state.transactions.some(t => t.ref === ref);
      if(!isDuplicate){
        state.transactions.unshift({
          id: crypto.randomUUID ? crypto.randomUUID() : String(Date.now()),
          type: type,
          account: account,
          amount: amount,
          commission: 0,
          deltas: txDeltas(type, amount, account),
          date: dateNow(),
          ref: ref,
          note: 'Auto SMS: ' + sms,
          created: new Date().toISOString()
        });
        save();
      }
    }
    window.history.replaceState({}, document.title, window.location.pathname);
  }
}

processAutoSMS();
render();
